import { Keyring } from '@polkadot/keyring';
import { cryptoWaitReady } from '@polkadot/util-crypto';
import { hexToU8a, u8aToHex } from '@polkadot/util';
import type { WalletConfig } from './types.js';
 
/**
 * PolkadotSigner interface expected by PAPI
 */
export interface PolkadotSigner {
  publicKey: Uint8Array;
  sign: (
    callData: Uint8Array,
    signedExtensions: Record<string, Uint8Array>,
    metadata: Uint8Array,
  ) => Promise<Uint8Array>;
}
 
/**
 * Create a signer from wallet configuration
 * Supports both private key and mnemonic (from env or config)
 */
export async function createSigner(config?: WalletConfig): Promise<PolkadotSigner> {
  // Wait for crypto to be ready
  await cryptoWaitReady();
 
  // Try to get private key from environment first
  const privateKeyEnv = process.env.WALLET_PRIVATE_KEY;
  const mnemonicEnv = process.env.WALLET_MNEMONIC;
 
  let privateKey: string | undefined;
  let mnemonic: string | undefined;
 
  // Priority: env vars > config
  if (privateKeyEnv) {
    privateKey = privateKeyEnv;
  } else if (mnemonicEnv) {
    mnemonic = mnemonicEnv;
  } else if (config?.type === 'private_key' && config.private_key) {
    privateKey = config.private_key;
  } else if (config?.type === 'mnemonic' && config.mnemonic) {
    mnemonic = config.mnemonic;
  } else {
    throw new Error(
      'No wallet configured. Set WALLET_PRIVATE_KEY or WALLET_MNEMONIC environment variable, ' +
      'or configure wallet in config.yaml'
    );
  }
 
  const keyring = new Keyring({ type: 'sr25519' });
 
  let keypair;
  if (privateKey) {
    // Handle private key (with or without 0x prefix)
    const cleanKey = privateKey.startsWith('0x') ? privateKey : `0x${privateKey}`;
    keypair = keyring.addFromSeed(hexToU8a(cleanKey));
  } else if (mnemonic) {
    // Handle mnemonic with optional derivation path
    const derivationPath = config?.derivation_path || '//0';
    keypair = keyring.addFromUri(`${mnemonic}${derivationPath}`);
  } else E{
    throw new Error('Invalid wallet configuration');
  }
 
  // Create PolkadotSigner
  return {
    publicKey: keypair.publicKey,
    sign: async (
      callData: Uint8Array,
      signedExtensions: Record<string, Uint8Array>,
      metadata: Uint8Array,
    ) => {
      // PAPI expects us to sign the payload
      // The payload construction is handled by PAPI, we just sign what we're given
 
      // Concatenate all data that needs to be signed
      const extensionsData = Object.values(signedExtensions);
      const allData = new Uint8Array(
        callData.length +
        extensionsData.reduce((acc, ext) => acc + ext.length, 0)
      );
 
      let offset = 0;
      allData.set(callData, offset);
      offset += callData.length;
 
      for (const ext of extensionsData) {
        allData.set(ext, offset);
        offset += ext.length;
      }
 
      // Sign the data
      const signature = keypair.sign(allData);
 
      return signature;
    },
  };
}
 
/**
 * Get the SS58 address for a signer
 */
export function getSignerAddress(signer: PolkadotSigner, ss58Format: number = 42): string {
  const keyring = new Keyring({ type: 'sr25519', ss58Format });
  const keypair = keyring.addFromSeed(signer.publicKey);
  return keypair.address;
}
